# Remove drop-shipping spam from new accounts
type: submission
title: ['look what I got', 'just arrived', 'grab yours here']
account_age: < 30
action: remove
modmail: "User {{author}}'s post was removed for suspected drop-shipping spam (new account). Link: {{permalink}}"
